+    /// Split a (potentially long) string into chunks that each fit into the available header
+    /// content width. The chunks are returned as a `Vec` of `String`s. If the string fits into a
+    /// single line, a single chunk is returned.
+    ///
+    /// `first_line_prefix` is the (unwrapped) text that is printed before the value on the first
+    /// line (e.g. `"File: "`). It is taken into account so that the first chunk is short enough
+    /// to leave room for it. `last_line_suffix` is the (unwrapped) text that is printed after the
+    /// value on the last line (e.g. `"   <BINARY>"`). It is taken into account so that the last
+    /// chunk is short enough to leave room for it.
+    fn split_into_header_chunks(
+        &self,
+        text: &str,
+        first_line_prefix: &str,
+        last_line_suffix: &str,
+    ) -> Vec<String> {
+        let available_width = match self
+            .config
+            .term_width
+            .checked_sub(self.header_component_indent_width())
+        {
+            Some(width) if width > 0 => width,
+            // If there is no space for any content, fall back to a single chunk to avoid
+            // underflow issues. The text will then simply overflow, which matches the previous
+            // behavior.
+            _ => return vec![text.to_string()],
+        };
+
+        let prefix_width = display_width(first_line_prefix);
+        let suffix_width = display_width(last_line_suffix);
+
+        let mut chunks: Vec<String> = Vec::new();
+        let mut current_chunk = String::new();
+        let mut current_width = 0usize;
+
+        for c in text.chars() {
+            let char_width = c.width().unwrap_or(0);
+
+            // The first chunk has to leave room for the prefix. Since we don't know in advance
+            // how many chunks we will end up with, we conservatively leave room for the suffix
+            // on every chunk (the suffix is only printed on the last line, but reserving space
+            // for it on every line guarantees that no line ever overflows).
+            let reserved_width = if chunks.is_empty() {
+                prefix_width + suffix_width
+            } else {
+                suffix_width
+            };
+
+            if current_width > 0 && current_width + char_width + reserved_width > available_width {
+                // The current chunk is full, push it and start a new one.
+                chunks.push(std::mem::take(&mut current_chunk));
+                current_width = 0;
+            }
+
+            current_chunk.push(c);
+            current_width += char_width;
+        }
+
+        if !current_chunk.is_empty() || chunks.is_empty() {
+            chunks.push(current_chunk);
+        }
+
+        chunks
+    }
