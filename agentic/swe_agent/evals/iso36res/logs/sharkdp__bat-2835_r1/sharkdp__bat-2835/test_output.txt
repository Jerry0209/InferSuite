+    fn print_header_multiline_component(
+        &mut self,
+        handle: &mut OutputHandle,
+        content: &str,
+    ) -> Result<()> {
+        let content_width = self.config.term_width - self.get_header_component_indent_length();
+
+        // The content may contain ANSI escape sequences (e.g. when the file name is
+        // colored) as well as multi-byte characters. We therefore can't simply split
+        // the string at an arbitrary byte index, as that could either panic (when
+        // splitting in the middle of a character) or garble escape sequences. Instead,
+        // we iterate over the content, keeping track of the displayed width, and wrap
+        // at character boundaries whenever a line would exceed the available width.
+        let mut current_line = String::new();
+        let mut current_width = 0usize;
+
+        for (chunk, is_ansi) in AnsiCodeIterator::new(content) {
+            if is_ansi {
+                // Escape sequences don't take up any visible width, but we still need
+                // to emit them so that coloring is preserved.
+                current_line.push_str(chunk);
+                continue;
+            }
+
+            for c in chunk.chars() {
+                let char_width = c.width().unwrap_or(0);
+
+                if current_width + char_width > content_width && !current_line.is_empty() {
+                    self.print_header_component_with_indent(handle, &current_line)?;
+                    current_line.clear();
+                    current_width = 0;
+                }
+
+                current_line.push(c);
+                current_width += char_width;
+            }
+        }
+
+        self.print_header_component_with_indent(handle, &current_line)
+    }
