+    fn paging_mode(&self, inputs: &[Input]) -> PagingMode {
+        // The paging behavior can be controlled via three different (possibly
+        // conflicting) command-line options:
+        //   * `--paging=auto|never|always`
+        //   * `-P` / `--no-paging` (alias for `--paging=never`)
+        //   * `-pp` (i.e. `-p` specified twice, alias for `--paging=never`)
+        //
+        // When several of these are specified (e.g. `--paging=always` in the
+        // config file and `-P` on the command line), the one that appears *last*
+        // on the (effective) command line takes precedence. We determine the
+        // winner by comparing the clap indices of the relevant options.
+        //
+        // See https://github.com/sharkdp/bat/issues/2264
+
+        // The index of the last occurrence of `--paging`, but only if it was
+        // explicitly specified by the user (otherwise `index_of` would return the
+        // index of the injected default value, which always comes last).
+        let paging_index = if self.matches.occurrences_of("paging") > 0 {
+            self.matches
+                .indices_of("paging")
+                .and_then(|mut indices| indices.next_back())
+        } else {
+            None
+        };
+
+        // The index of the last `-P` / `--no-paging`.
+        let no_paging_index = self
+            .matches
+            .indices_of("no-paging")
+            .and_then(|mut indices| indices.next_back());
+
+        // The index at which `-p` disables paging, i.e. the index of the *second*
+        // `--plain`/`-p` occurrence. Only relevant if `-p` was given at least twice.
+        let plain_disable_index = if self.matches.occurrences_of("plain") > 1 {
+            self.matches
+                .indices_of("plain")
+                .and_then(|mut indices| indices.nth(1))
+        } else {
+            None
+        };
+
+        // If `--paging` was explicitly specified and it appears *after* every
+        // `-P`/`-pp`, its value wins.
+        let paging_overrides_short_options = paging_index.map_or(false, |paging_index| {
+            paging_index >= no_paging_index.unwrap_or(0)
+                && paging_index >= plain_disable_index.unwrap_or(0)
+        });
+
+        if paging_overrides_short_options {
+            return match self.matches.value_of("paging") {
+                Some("always") => PagingMode::Always,
+                Some("never") => PagingMode::Never,
+                Some("auto") => self.paging_mode_auto(inputs),
+                _ => unreachable!("other values for --paging are not allowed"),
+            };
+        }
+
+        // Otherwise, `-P` or `-pp` (whichever comes last) disables the pager, as
+        // long as one of them was specified at all.
+        if no_paging_index.is_some() || plain_disable_index.is_some() {
+            return PagingMode::Never;
+        }
+
+        // Neither `--paging` nor `-P`/`-pp` were specified, fall back to the
+        // automatic behavior.
+        self.paging_mode_auto(inputs)
+    }
