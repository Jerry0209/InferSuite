+    fn paging_mode(&self, inputs: &[Input]) -> PagingMode {
+        // The paging behavior can be controlled via three (mutually overridable) sources:
+        //   * `--paging=...` (`auto`, `never` or `always`)
+        //   * `-pp` (an alias for `--paging=never`)
+        //   * `-P` / `--no-paging` (an alias for `--paging=never`)
+        //
+        // When several of these are specified, the one given *last* on the command line
+        // takes precedence (POSIX-style "last wins"). We determine the winner by comparing
+        // the positions (indices) at which the arguments occurred.
+        //
+        // Note that a single `-p`/`--plain` only affects the *style* and must not influence
+        // paging, so `-p`/`--plain` is only treated as a paging argument when it has been
+        // specified at least twice (`-pp`).
+
+        // The index of `--paging=...`, but only if it was explicitly provided on the command
+        // line (i.e. not just the default value).
+        let paging_index = (self.matches.value_source("paging") == Some(ValueSource::CommandLine))
+            .then(|| self.matches.index_of("paging"))
+            .flatten();
+
+        // The index of `-pp` (only relevant when `--plain` has been specified at least twice).
+        let extra_plain_index = (self.matches.get_count("plain") > 1)
+            .then(|| self.matches.index_of("plain"))
+            .flatten();
+
+        // The index of `-P` / `--no-paging`.
+        let no_paging_index = self
+            .matches
+            .get_flag("no-paging")
+            .then(|| self.matches.index_of("no-paging"))
+            .flatten();
+
+        // Whichever paging-related argument was specified last "wins".
+        let paging_arg = [
+            paging_index.map(|index| (index, "paging")),
+            extra_plain_index.map(|index| (index, "extra-plain")),
+            no_paging_index.map(|index| (index, "no-paging")),
+        ]
+        .into_iter()
+        .flatten()
+        .max_by_key(|(index, _)| *index)
+        .map(|(_, arg)| arg);
+
+        match paging_arg {
+            // `--paging=...` was specified last (or is the only source). Use its value.
+            Some("paging") => match self.matches.get_one::<String>("paging").map(|s| s.as_str()) {
+                Some("always") => PagingMode::Always,
+                Some("never") => PagingMode::Never,
+                Some("auto") | None => self.paging_mode_auto(inputs),
+                _ => unreachable!("other values for --paging are not allowed"),
+            },
+            // `-pp` or `-P` was specified last: paging is disabled.
+            Some("extra-plain") | Some("no-paging") => PagingMode::Never,
+            // None of the paging-related arguments was specified: use the automatic behavior.
+            None => self.paging_mode_auto(inputs),
+            _ => unreachable!("other values for paging argument are not allowed"),
